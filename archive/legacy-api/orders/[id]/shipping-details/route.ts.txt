import { NextResponse } from 'next/server';
import { createClient } from '@/utils/supabase/server';

export async function GET(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const orderId = (await params).id;
  const supabase = await createClient();

  // 1. Get the current user from the session
  const { data: { user }, error: authError } = await supabase.auth.getUser();

  if (authError || !user) {
    return new NextResponse(JSON.stringify({ error: 'Unauthorized' }), { status: 401 });
  }

  // 2. Fetch the order directly with seller_id
  const { data: order, error: orderError } = await supabase
    .from('orders')
    .select('shipping_address_id, seller_id')
    .eq('id', orderId)
    .single();

  if (orderError) {
    console.error('Error fetching order:', orderError);
    return new NextResponse(JSON.stringify({ error: 'Order not found' }), { status: 404 });
  }

  // 3. Verify that the current user is the seller
  if (!order.seller_id || order.seller_id !== user.id) {
    return new NextResponse(JSON.stringify({ error: 'Forbidden' }), { status: 403 });
  }

  // 4. Check if shipping address ID exists
  if (!order.shipping_address_id) {
    return new NextResponse(JSON.stringify({
      error: 'No shipping address found for this order. The order may not have been paid yet.'
    }), { status: 404 });
  }

  // 5. Fetch the shipping address using the ID from the order
  const { data: shippingAddress, error: addressError } = await supabase
    .from('shipping_addresses')
    .select('*')
    .eq('id', order.shipping_address_id)
    .single();

  if (addressError) {
    console.error('Error fetching shipping address:', addressError);
    return new NextResponse(JSON.stringify({ error: 'Shipping address not found' }), { status: 404 });
  }

  // 6. Return the shipping address
  return NextResponse.json(shippingAddress);
}
