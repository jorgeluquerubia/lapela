import { NextResponse } from 'next/server';
import { createClient } from '@/utils/supabase/server';

export async function POST(request: Request) {
  const supabase = await createClient();

  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    return new NextResponse('Unauthorized', { status: 401 });
  }

  const body = await request.json();
  const { product_id, seller_id, shipping_address_id } = body;

  if (!product_id || !seller_id || !shipping_address_id) {
    return new NextResponse('Missing required fields', { status: 400 });
  }

  console.log('Finalizing purchase for product:', product_id);
  console.log('User:', user.id);
  console.log('Shipping address:', shipping_address_id);

  // Find the existing pending_details order instead of creating a new one
  const { data: existingOrders, error: findError } = await supabase
    .from('orders')
    .select('*')
    .eq('product_id', product_id)
    .eq('buyer_id', user.id)
    .eq('status', 'pending_details')
    .order('created_at', { ascending: false });

  if (findError) {
    console.error('Error finding existing order:', findError);
    return new NextResponse(JSON.stringify({ error: 'Could not find pending order' }), { status: 500 });
  }

  if (!existingOrders || existingOrders.length === 0) {
    console.error('No pending payment order found for product:', product_id);
    return new NextResponse(JSON.stringify({ error: 'No pending order found for this product' }), { status: 404 });
  }

  // Use the most recent pending order
  const existingOrder = existingOrders[0];
  console.log('Found existing order:', existingOrder.id);

  // Update the existing order with shipping details
  const { data: updatedOrder, error: updateError } = await supabase
    .from('orders')
    .update({
      shipping_address_id,
      status: 'pending_payment', // The user has added address and is now pending payment
      updated_at: new Date().toISOString()
    })
    .eq('id', existingOrder.id)
    .select()
    .single();

  if (updateError) {
    console.error('Error updating order:', updateError);
    return new NextResponse(JSON.stringify({ error: 'Could not finalize the purchase' }), { status: 500 });
  }

  console.log('Order updated successfully:', updatedOrder.id);

  // Update the product status to 'awaiting_payment'
  const { error: productError } = await supabase
    .from('products')
    .update({ status: 'awaiting_payment' })
    .eq('id', product_id);

  if (productError) {
    console.error('Error updating product status:', productError);
    // Try to revert the order update if product update fails
    await supabase
      .from('orders')
      .update({ status: 'pending_details' })
      .eq('id', existingOrder.id);
    return new NextResponse(JSON.stringify({ error: 'Could not complete the sale process' }), { status: 500 });
  }

  console.log('✅ Purchase finalized successfully');
  console.log('- Order ID:', updatedOrder.id);
  console.log('- Product status: awaiting_payment');
  console.log('- Order status: pending_payment');

  return NextResponse.json(updatedOrder);
}
